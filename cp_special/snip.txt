    int n, m;

    boolean check (int at) {
        ST st = new ST(0, n, 0);
        for(int i = 0; i <= at; i++) {
            st.set(qq[i], 1);
        }
        for(int[] rn: rng) {
            int l = rn[0], r = rn[1];
            int ones = st.query(l, r);
            int zeroes = r-l+1-ones;
            if(ones > zeroes) {
                return true;
            }
        }
        return false;
    }

    void solve() {
        n = ni();
        m = ni();
        rng = new int[m][];
        for(int i = 0; i < m; i++) rng[i] = new int[]{ni()-1, ni()-1};
        int q = ni();
        qq = new int[q];
        for(int i = 0; i < q; i++) qq[i] = ni()-1;
        int l = -1; // false
        int r = q; // true
        while(l < r - 1) {
            int mid = (l+r)>>1;
            if(check(mid)) r = mid;
            else l = mid;
        }
        if(r == q) {
            p(-1);
        } else p((r+1));
    }

    // Ultra Fast Segment Tree inspired from Tlatoani's code
    // https://codeforces.com/contest/1208/submission/59470935
    // [l, r]
    class ST {
        final int[] val;
        final int treeFrom;
        final int length;
        final int neutralElem;

        int func (int a, int b) {
            return a+b;
        }

        ST(int treeFrom, int treeTo, int neutralElem) {
            this.neutralElem = neutralElem;
            this.treeFrom = treeFrom;
            int length = treeTo - treeFrom + 1;
            int l = Integer.highestOneBit(length);
            if(length > l) l <<= 1;
            val = new int[l << 1];
            this.length = l;
            // Important to fill the tree with neutral values
            Arrays.fill(val, neutralElem);
        }

        ST(int[] arr, int neutralElem) {
            this.neutralElem = neutralElem;
            int length = arr.length;
            this.treeFrom = 0;
            int l = Integer.highestOneBit(length);
            if(length > l) l <<= 1;
            val = new int[l << 1];
            this.length = l;
            // Important to fill the tree with neutral values
            Arrays.fill(val, neutralElem);

            for(int i = 0; i < length; i++) {
                set(i, arr[i]);
            }
        }

        void increase(int index, int delta) {
            int node = index - treeFrom + length; // gives the index of leaf
            val[node] += delta;
            // going upwards from leaf
            for (node >>= 1; node > 0; node >>= 1) {
                val[node] = func(val[node << 1], val[(node << 1) + 1]);
            }
        }

        void set(int index, int value) {
            int node = index - treeFrom + length; // gives the index of the leaf
            val[node] = value;
            // going upwards from leaf
            for (node >>= 1; node > 0; node >>= 1) {
                val[node] = func(val[node << 1], val[(node << 1) + 1]);
            }
        }

        int query(int from, int to) {
            from += length - treeFrom; // index of left leaf
            to += length - treeFrom + 1; // index of right leaf

            int res = neutralElem;

            for (; from + (from & -from) <= to; from += from & -from) {
                res = func(res, val[from / (from & -from)]);
            }

            for (; to - (to & -to) >= from; to -= to & -to) {
                res = func(val[(to - (to & -to)) / (to & -to)], res);
            }

            return res;
        }
    }
